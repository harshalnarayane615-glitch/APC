# Q7: Single Python Program with Output

import pandas as pd
import os

# a) Copy a file after converting its contents to uppercase

with open("input.txt", "w") as f:
    f.write("welcome to python programming")

with open("input.txt", "r") as f:
    data = f.read()

with open("output.txt", "w") as f:
    f.write(data.upper())

print("A) File copied with uppercase content.")
print("Output:", data.upper())
